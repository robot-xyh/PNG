# Betaflight 2025.12.2 SITL configuration matching the MICOAIR743V2 control contract.
feature GPS
# Betaflight 2025.12.2 config validation requires a declared GPS UART even
# when the SITL-only VIRTUAL provider consumes FDM data instead of serial data.
serial UART2 2 115200 57600 0 115200
set gps_provider = VIRTUAL
# SITL's serial task processes at most one MSP frame per invocation. The
# runner schedules ~46 Hz telemetry plus 50 Hz RC, so 100 Hz has no margin.
set serial_update_rate_hz = 2000
aux 0 0 0 900 1300 0 0
aux 1 1 3 1375 2100 0 0
aux 2 50 2 1700 2100 0 0
set msp_override_channels_mask = 15
set msp_override_failsafe = OFF
set p_pitch = 54
set i_pitch = 67
set d_pitch = 25
set p_roll = 51
set i_roll = 64
set d_roll = 22
set roll_rc_rate = 100
set pitch_rc_rate = 100
set roll_srate = 70
set pitch_srate = 70
set roll_expo = 0
set pitch_expo = 0
